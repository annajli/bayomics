# bnlearn's C backend does NOT support tibbles. A tibble keeps its tbl_df class
# through subsetting, which makes bn.fit misread factor columns and throw
# "REAL() can only be applied to a 'numeric', not a 'integer'" on conditional-
# Gaussian nodes. Coerce to a plain data.frame before any bnlearn call.
bn_df <- as.data.frame(bn_df)

# Belt-and-suspenders: ensure continuous columns are double, not integer.
for (col in setdiff(names(bn_df), ROOTS)) {
  bn_df[[col]] <- as.numeric(bn_df[[col]])
}

2. Generate blacklist

No edges into root nodes + roots mutually independent. CLG constraint: discrete nodes cannot have continuous parents.

all_nodes <- names(bn_df)
non_roots <- setdiff(all_nodes, ROOTS)

# Rule 1: No edges INTO root nodes (from any node)
bl_into_roots <- expand.grid(from = all_nodes, to = ROOTS, stringsAsFactors = FALSE) %>%
  filter(from != to)

# Rule 2: Roots mutually independent (no edges between roots in either direction)
# Already covered by Rule 1 (roots are in the "to" set), but make explicit:
bl_between_roots <- expand.grid(from = ROOTS, to = ROOTS, stringsAsFactors = FALSE) %>%
  filter(from != to)

# Rule 3: Block edges between WBC and its deterministic components
# (WBC ≈ sum of the differential counts — same collinearity issue as the
# already-excluded NLR variable, but here we keep WBC as a node since it may
# carry non-redundant relationships to other markers, e.g. CRP/ESR).
# Filtered to intersect(all_nodes) since not every network config includes
# wbc or all 5 differential counts (e.g. L1b's curated panel omits basophils) —
# referencing a blacklist node that doesn't exist in bn_df throws an error in
# build.blacklist(), so this must stay defensive across all three configs.
wbc_components <- c("neutrophils", "lymphocytes", "monocytes", "eosinophils", "basophils")
wbc_cluster <- intersect(c("wbc", wbc_components), all_nodes)

if (length(wbc_cluster) >= 2) {
  bl_wbc <- expand.grid(from = wbc_cluster, to = wbc_cluster, stringsAsFactors = FALSE) %>%
    filter(from != to)
  cat(sprintf("WBC cluster present: %s (%d blacklist rows)\n",
              paste(wbc_cluster, collapse = ", "), nrow(bl_wbc)))
} else {
  bl_wbc <- data.frame(from = character(0), to = character(0))
  cat("WBC cluster: fewer than 2 relevant nodes present, no rule applied\n")
}
## WBC cluster present: wbc, neutrophils, lymphocytes, monocytes, eosinophils (20 blacklist rows)
# Rule 4: Block direct parent-child edges in hierarchical modalities (e.g. freq's
# L1/L2/L3 cell-type gating tree). Only applies if the current config defines a
# PARENT_MAP (freq does; clinical/Olink/wb/pb don't) -- defensively checked with
# exists() so this safely no-ops for every other network, the same way the WBC
# rule's intersect() no-ops when wbc_cluster isn't relevant.
if (exists("PARENT_MAP")) {
  child_nodes <- intersect(names(PARENT_MAP), all_nodes)
  bl_hierarchy_rows <- lapply(child_nodes, function(child) {
    parent <- PARENT_MAP[[child]]
    if (parent %in% all_nodes) {
      data.frame(from = c(child, parent), to = c(parent, child), stringsAsFactors = FALSE)
    } else {
      NULL
    }
  })
  bl_hierarchy <- bind_rows(bl_hierarchy_rows)
  cat(sprintf("Hierarchy blacklist (PARENT_MAP): %d parent-child pairs, %d blacklist rows\n",
              nrow(bl_hierarchy) / 2, nrow(bl_hierarchy)))
} else {
  bl_hierarchy <- data.frame(from = character(0), to = character(0))
}

blacklist <- rbind(bl_into_roots, bl_between_roots, bl_wbc, bl_hierarchy) %>% distinct()
cat(sprintf("Blacklist: %d edges forbidden\n", nrow(blacklist)))
## Blacklist: 92 edges forbidden

5a. Averaged networks at multiple thresholds

# Optimal threshold (bnlearn heuristic)
avg_opt <- averaged.network(boot_str)
thresh_opt <- attr(avg_opt, "threshold")
if (is.null(thresh_opt) || is.na(thresh_opt)) {
  cat("Optimal threshold heuristic returned NA/NULL — falling back to 0.5\n")
  thresh_opt <- 0.5
  avg_opt <- averaged.network(boot_str, threshold = thresh_opt)
}
## Optimal threshold heuristic returned NA/NULL — falling back to 0.5
cat(sprintf("Optimal threshold: %.4f\n", thresh_opt))
## Optimal threshold: 0.5000
cat(sprintf("Averaged network (optimal, t=%.4f): %d edges\n", thresh_opt, narcs(avg_opt)))
## Averaged network (optimal, t=0.5000): 36 edges
# Conservative threshold — empirical valley in mid-range strengths (0.3-0.95).
# Uses binned histogram rather than density()/KDE: a smoothed density estimate
# can be pulled toward whichever boundary is closest to the near-1.0 cluster
# (confirmed on L1a, where KDE reported 0.95 — right at the search boundary —
# while the real minimum-count bin was 0.80-0.85). Binning can't overshoot.
# Fallback to 0.85 when too few mid-range edges exist to estimate a valley.
strengths_mid <- boot_str$strength[boot_str$strength > 0.3 & boot_str$strength < 0.95]
if (length(strengths_mid) >= 5) {
  bin_width <- 0.05
  breaks <- seq(0.3, 0.95, by = bin_width)
  h <- hist(strengths_mid, breaks = breaks, plot = FALSE)
  min_bin <- which.min(h$counts)
  thresh_conservative <- h$mids[min_bin]
  cat(sprintf("Bin counts (0.3-0.95, width %.2f): %s\n",
              bin_width, paste(h$counts, collapse = ", ")))
} else {
  cat("Too few mid-range edge strengths to estimate a valley — falling back to 0.85\n")
  thresh_conservative <- 0.85
}
## Bin counts (0.3-0.95, width 0.05): 48, 34, 28, 10, 20, 12, 8, 2, 10, 4, 2, 4, 2
cat(sprintf("Conservative threshold (empirical valley): %.3f\n", thresh_conservative))
## Conservative threshold (empirical valley): 0.675
avg_conservative <- averaged.network(boot_str, threshold = thresh_conservative)
cat(sprintf("Averaged network (conservative, t=%.4f): %d edges\n", thresh_conservative, narcs(avg_conservative)))
## Averaged network (conservative, t=0.6750): 15 edges
# Liberal threshold (0.50)
